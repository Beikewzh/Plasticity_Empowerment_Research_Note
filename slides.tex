\begin{frame}[shrink=6]{Setting}
\begin{center}
\begin{tikzpicture}[>={Stealth[length=2.5mm]}, node distance=14mm and 24mm,
  box/.style={draw, rounded corners=3pt, minimum height=11mm, minimum width=30mm, align=center, font=\small}]
\node[box] (world) {environment\\ $e(O_{t+1}\mid h_t, A_t)$};
\node[box, right=of world] (mem) {memory\\ $M_t = f(M_{t-1}, A_{t-1}, O_t)$};
\node[box, right=of mem] (pol) {policy\\ $A_t \sim \pi(\cdot \mid M_t)$};
\draw[->, thick] (world) -- node[above, font=\small] {$O_t$} (mem);
\draw[->, thick] (mem) -- node[above, font=\small] {$M_t$} (pol);
\draw[->, thick] (pol.south) -- ++(0,-9mm) -| node[pos=0.25, below, font=\small] {$A_t$} (world.south);
\end{tikzpicture}
\end{center}
\vspace{2mm}
\begin{itemize}
\item History $h_t = (O_{1:t}, A_{1:t-1})$. The reward $R_t$ is one component of the observation $O_t$.
\item The environment may depend on the whole history; no Markov assumption. A changing environment is one such law.
\item The memory is a deterministic function of the history, $M_t = \phi_t(h_t)$, and the action depends on $h_t$ only through $M_t$.
\item Finite observation and action sets, finite horizon $T$. All probabilities are under the distribution induced by the three boxes.
\end{itemize}
\vspace{2mm}
Question: when does the memory contain everything needed to predict the consequences of actions?
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Definition: state defect}
\begin{block}{}
\[
D_t \;=\; I\big(h_t;\, O_{t+1} \,\big|\, M_t, A_t\big)
\]
\end{block}
\vspace{1mm}
Given the memory and the action, how much information about the next observation is still contained in the history?
\vspace{3mm}
\begin{itemize}
\item $D_t = 0$ if and only if any two histories with the same memory value give the same distribution of $O_{t+1}$, for every action the agent takes there.
\item If $D_t = 0$ for all $t$, we call $M$ a \textbf{sufficient predictive state}.
\item The condition concerns the pairs $(h_t, a)$ that occur under the agent's own behaviour, not all histories and all actions.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Sufficiency for control}
\begin{enumerate}
\item The reward is a component of $O_{t+1}$, and $M_{t+1} = f(M_t, A_t, O_{t+1})$. If $O_{t+1}$ is independent of $h_t$ given $(M_t, A_t)$, so is any function of $(M_t, A_t, O_{t+1})$:
  \[
  \Pr(R_{t+1}, M_{t+1} \mid h_t, A_t) = \Pr(R_{t+1}, M_{t+1} \mid M_t, A_t) .
  \]
\item Hence (reward, next memory) evolves as a Markov decision process on memory values, and the Bellman recursion can be written on memory, time by time:
  \[
  q^*_t(m, a) = \sum_{r, m'} \Pr(r, m' \mid m, a)\,[\,r + \gamma\, v^*_{t+1}(m')\,], \qquad v^*_t(m) = \max_{a:\ \pi_t(a \mid m) > 0} q^*_t(m, a).
  \]
\item $v^*_t(M_t)$ equals the best return over all behaviours, history-dependent or not, that use only actions the agent takes. If every action has positive probability, $q^*_t(M_t, a) = Q^*_t(h_t, a)$ and the greedy policy is optimal.
\end{enumerate}
\vspace{2mm}
This is an existence statement: an optimal policy that reads $M_t$ exists. Whether a learning algorithm finds it is a separate question.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}[shrink=8]{Example: matching pennies}
\begin{itemize}\setlength{\itemsep}{6pt}
\item Each step a fair coin is shown. The agent guesses heads or tails and is paid $1$ if the guess matches the coin.
\item Suppose the agent remembers nothing from one step to the next, and guesses at random.
\item Ask it to predict what it will see next. A fresh coin: fifty--fifty. A reward: also fifty--fifty, because its guess is random. It predicts this perfectly, and knowing the whole past would not improve the prediction at all.
\item Now ask it: ``if you guess heads right now, what will the reward be?'' It cannot say. The answer depends on the coin it has already forgotten. An agent that remembered the coin would answer with certainty.
\item So this memory passes every prediction test and still cannot support a better policy. What it lacks is knowledge of what its own action does: the empowerment it has lost.
\end{itemize}
\vspace{3mm}
The next slides make this exact: the gap between ``predicts the future'' and ``is a sufficient state'' is precisely the empowerment the memory fails to see. The numbers for this example are on the last slide.
\end{frame}

\begin{frame}{Plasticity and empowerment, at the history and at the memory}
Abel et al.\ (2025) define, in one-step form,
\begin{columns}[T]
\column{0.5\textwidth}
empowerment, the action's effect on the next observation,
\[
E_t = I(A_t;\, O_{t+1} \mid h_t),
\]
\column{0.5\textwidth}
plasticity, the observation's effect on the next action,
\[
P_{t+1} = I(O_{t+1};\, A_{t+1} \mid h_t, A_t).
\]
\end{columns}
\vspace{3mm}
The agent cannot condition on the full history. The same quantities computed from the memory are
\[
E_t(M) = I(A_t;\, O_{t+1} \mid M_t), \qquad P_{t+1}(M) = I(O_{t+1};\, A_{t+1} \mid M_t, A_t).
\]
\vspace{1mm}
We study the two mismatches
\[
\dE_t = E_t - E_t(M), \qquad \dP_t = P_{t+1}(M) - P_{t+1},
\]
not the absolute values of $E$ or $P$.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Empowerment mismatch}
Each mutual information is a difference of conditional entropies:
\begin{align*}
E_t &= H(A_t \mid h_t) \;-\; H(A_t \mid h_t, O_{t+1}) \\
E_t(M) &= H(A_t \mid M_t) \;-\; H(A_t \mid M_t, O_{t+1})
\end{align*}
The action is drawn from $\pi(\cdot \mid M_t)$, so given $M_t$ it is independent of $h_t$:
\[
H(A_t \mid h_t) = H(A_t \mid M_t).
\]
Subtracting, the first terms cancel:
\[
\dE_t = H(A_t \mid M_t, O_{t+1}) - H(A_t \mid h_t, O_{t+1}).
\]
Since $M_t$ is a function of $h_t$, $H(A_t \mid h_t, O_{t+1}) = H(A_t \mid h_t, M_t, O_{t+1})$, and a difference $H(X \mid Z) - H(X \mid Y, Z)$ is the mutual information $I(Y; X \mid Z)$:
\[
\boxed{\;\dE_t = I(h_t;\, A_t \mid M_t, O_{t+1}) \;\ge\; 0\;}
\]
Interpretation: after the outcome is observed, the discarded part of the history still carries information about the action. The memory can only under-estimate empowerment.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Plasticity mismatch}
\begin{align*}
P_{t+1} &= H(A_{t+1} \mid h_t, A_t) \;-\; H(A_{t+1} \mid h_t, A_t, O_{t+1}) \\
P_{t+1}(M) &= H(A_{t+1} \mid M_t, A_t) \;-\; H(A_{t+1} \mid M_t, A_t, O_{t+1})
\end{align*}
The two second terms are equal:
\begin{itemize}
\item $(h_t, A_t, O_{t+1}) = h_{t+1}$, and $A_{t+1}$ depends on $h_{t+1}$ only through $M_{t+1}$, so the first is $H(A_{t+1} \mid M_{t+1})$;
\item $(M_t, A_t, O_{t+1})$ determines $M_{t+1} = f(M_t, A_t, O_{t+1})$, and $A_{t+1}$ depends only on $M_{t+1}$, so the second is also $H(A_{t+1} \mid M_{t+1})$.
\end{itemize}
Subtracting, these cancel:
\begin{align*}
\dP_t \;=\; P_{t+1}(M) - P_{t+1} &\;=\; H(A_{t+1} \mid M_t, A_t) - H(A_{t+1} \mid h_t, A_t) \\
&\;=\; \boxed{\; I(h_t;\, A_{t+1} \mid M_t, A_t) \;\ge\; 0 \;}
\end{align*}
Interpretation: the discarded part of the history still predicts the agent's next action. The memory can only over-estimate plasticity.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Two facts about mutual information used next}
\textbf{Chain rule.} For any random variables $X, Y, Z, W$, the information that the pair $(X, Y)$ carries about $Z$ given $W$ can be accumulated in either order:
\[
I(X, Y;\, Z \mid W) \;=\; I(X;\, Z \mid W) + I(Y;\, Z \mid W, X) \;=\; I(Y;\, Z \mid W) + I(X;\, Z \mid W, Y).
\]
\vspace{1mm}
\textbf{Data processing.} If, given $W$, the variable $Z$ depends on $X$ only through $Y$ (a chain $X \to Y \to Z$), then
\[
I(X;\, Z \mid W) \;\le\; I(X;\, Y \mid W).
\]
\vspace{1mm}
\textbf{A convention.} When $M_t$ is a function of $h_t$, conditioning on $(h_t, M_t)$ is the same as conditioning on $h_t$.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Decomposition of the state defect}
Apply the chain rule to $I(h_t, A_t;\, O_{t+1} \mid M_t)$ with $X = h_t$, $Y = A_t$, $Z = O_{t+1}$, $W = M_t$.
\begin{align*}
\text{history first:}\quad I(h_t, A_t;\, O_{t+1} \mid M_t) &= \underbrace{I(h_t; O_{t+1} \mid M_t)}_{D^0_t} + \underbrace{I(A_t; O_{t+1} \mid M_t, h_t)}_{=\,I(A_t; O_{t+1} \mid h_t)\,=\,E_t} \\[1mm]
\text{action first:}\quad I(h_t, A_t;\, O_{t+1} \mid M_t) &= \underbrace{I(A_t; O_{t+1} \mid M_t)}_{E_t(M)} + \underbrace{I(h_t; O_{t+1} \mid M_t, A_t)}_{D_t}
\end{align*}
Here $D^0_t = I(h_t; O_{t+1} \mid M_t)$ is the defect of prediction that ignores the action. Equating, $D^0_t + E_t = E_t(M) + D_t$, so
\[
\boxed{\;D_t \;=\; D^0_t \;+\; \dE_t\;}
\]
For plasticity, given $(M_t, A_t)$ the variables form the chain $h_t \to O_{t+1} \to M_{t+1} \to A_{t+1}$; data processing gives
\[
\boxed{\;0 \;\le\; \dP_t \;\le\; D_t\;}
\]
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Consequences}
\begin{block}{Theorem}
$M$ is a sufficient predictive state ($D_t = 0$) if and only if $D^0_t = 0$ and $\dE_t = 0$: it predicts the next observation without the action as well as the history does, \emph{and} it loses no empowerment.
\end{block}
\vspace{2mm}
\begin{itemize}
\item Prediction of the observation stream alone is not sufficient. In the pennies example $D^0_t = 0$ while $D_t = 1$ bit; the missing bit is $\dE_t$, the empowerment the constant memory does not see.
\item The plasticity mismatch is not a third term. It is a lower bound: $\dP_t > 0$ proves that the memory is insufficient, and this is read from the agent's own actions without any model of the environment.
\item $\dP_t = 0$ proves nothing. Under a policy that ignores its memory (uniform, as in the pennies example), $\dP_t = 0$ for every memory while $D_t = 1$ bit.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}[shrink=6]{Plasticity under regime identifiability}
\textbf{Assumption.} Within a memory class $m$ and action $a$, the histories the memory has merged fall into finitely many \emph{regimes} $c$, each with its own next-observation law $p_c(\cdot \mid m, a)$; the policy's responses to them,
\[
K_{m,a}\, p_c, \qquad K_{m,a}(a' \mid o') = \pi_{t+1}\big(a' \mid f(m, a, o')\big) \ \ \text{(the policy composed with the memory update)},
\]
are linearly independent. In words: \emph{the agent behaves differently in different regimes}. This needs as many actions as regimes, not as many actions as observations.
\vspace{2mm}
\begin{block}{Theorem (sandwich)}
Let $\sigma$ be the smallest singular value of $K_{m,a}$ on the differences of regime laws and $\rho$ the smallest probability of a next observation in the class. Then, in nats,
\[
\Delta P_t \;\le\; D_t \;=\; D^0_t + \Delta E_t \;\le\; \frac{2}{\rho\,\sigma^2}\, \Delta P_t .
\]
\end{block}
Proof of the right side: $\mathrm{KL} \le \chi^2 \le \|p - \bar p\|_2^2/\rho \le \|K(p - \bar p)\|_2^2/(\rho\sigma^2) \le 2\,\mathrm{KL}(Kp \,\|\, K\bar p)/(\rho\sigma^2)$ (Pinsker), averaged over the class. The left side and the equality are unconditional.
\vspace{2mm}
\begin{itemize}
\item $\sigma$ measures how differently the agent acts across regimes; $K$ is the agent's own policy and memory update, so $\sigma$ is computable by the agent once the regime directions are known. $\rho$ is read from data.
\item Exact checks ($t = 2$): W1, policy reacts to the last reward: $\Delta P = 0.019 \le D = 0.120 \le 61 \times 0.019$. W11, policy plays the last reward: $0.29 \le 0.36 \le 4.3$. The constant is loose by a factor $5$ to $50$.
\item The assumption is needed: W11 with a random policy, and W3s with a policy that reacts only to the merged class, have $\sigma = 0$: $\Delta P_t = 0$ while $D_t > 0$. A policy that does not act on a distinction cannot reveal that the memory lost it.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}[shrink=5]{Finite-window approximation}
The identities above condition on the full history $h_t$, which is not available. During training the agent stores a window of its own trajectory,
\[
W_t = (O_{t-k:t}, A_{t-k:t-1}), \qquad M^+_t = (M_t, W_t),
\]
and compares predictions from $M_t$ with predictions from $M^+_t$:
\begin{align*}
L^*_O &= H(O_{t+1} \mid M_t) - H(O_{t+1} \mid M^+_t), \\
L^*_E &= H(A_t \mid M_t, O_{t+1}) - H(A_t \mid M^+_t, O_{t+1}), \\
L^*_P &= H(A_{t+1} \mid M_t, A_t) - H(A_{t+1} \mid M^+_t, A_t).
\end{align*}
The same chain-rule argument with $W_t$ in place of $h_t$ gives
\begin{align*}
L^*_O + L^*_E &= I(W_t;\, O_{t+1} \mid M_t, A_t) =: D^{(k)}_t, \qquad 0 \le L^*_P \le D^{(k)}_t \le D_t, \\
D_t - D^{(k)}_t &= I(h_t;\, O_{t+1} \mid M_t, A_t, W_t).
\end{align*}
The last term is the predictive information older than the window. Example: reward determined by the action taken three steps earlier, window of one step, constant memory: all three window gaps are $0$, but $D_t = 1$ bit.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}[shrink=8]{Estimating the quantities from data}
A conditional entropy is estimated with a predictor trained by maximum likelihood. For $H(O_{t+1} \mid M_t)$:
\begin{itemize}
\item train a network $q_1(o \mid M_t)$ to predict the next observation from the memory, by minimising the average of $-\log q_1(O_{t+1} \mid M_t)$ over the stored transitions;
\item this average is the cross-entropy; it is $\ge H(O_{t+1} \mid M_t)$, with equality when $q_1$ has learned the true conditional distribution.
\end{itemize}
For $H(O_{t+1} \mid M^+_t)$ train a second network $q_2(o \mid M_t, W_t)$ with the window as extra input. Then
\[
\hat L_O \;=\; \text{mean}\big[-\log q_1(O_{t+1} \mid M_t)\big] \;-\; \text{mean}\big[-\log q_2(O_{t+1} \mid M_t, W_t)\big] .
\]
The randomness of the environment appears in both terms and cancels; only what the window adds remains.
\vspace{2mm}
Likewise $\hat L_E$ from two inverse models $q(a \mid M_t, O_{t+1})$ and $q(a \mid M_t, W_t, O_{t+1})$, and $\hat L_P$ from two next-action predictors $q(a' \mid M_t, A_t)$ and $q(a' \mid M_t, W_t, A_t)$.
\vspace{2mm}
These are estimates: exact only when both networks have learned the true conditional laws. A weak second network biases the estimate toward zero, so a positive estimate is evidence of a real gap.
\vspace{2mm}
Cost. The memory itself is trained without any second network (the window networks add no gradient). The second networks are trained offline on the stored buffer, can be small, and can take as input the stored memory state from $k$ steps earlier together with the actions and observations since, which truncated backpropagation stores anyway.
\end{frame}

\begin{frame}[shrink=12]{Training procedure}
\begin{center}
\begin{tikzpicture}[>={Stealth[length=2mm]}, node distance=3mm and 10mm,
  box/.style={draw, rounded corners=3pt, minimum height=7mm, minimum width=24mm, align=center, font=\scriptsize},
  dim/.style={box, dashed, draw=black!60, text=black!60}]
\node[box] (mem) {memory $f_\theta$\\ $M_t$};
\node[box, above right=of mem] (ho) {predictor of $O_{t+1}$ from $M_t$\\ loss $\ell_O$};
\node[box, right=of mem] (he) {inverse model: $A_t$ from $M_t, O_{t+1}$\\ loss $\ell_E$};
\node[box, below right=of mem] (pi) {policy $\pi_\nu$, value $V_\nu$\\ PPO on $\sg[M_t]$};
\node[dim, right=of ho] (hop) {same predictor from $(M_t, W_t)$\\ estimate $\hat L_O$};
\node[dim, right=of he] (hep) {same inverse model from $(M_t, W_t)$\\ estimate $\hat L_E$};
\node[dim, right=of pi] (hp) {next-action predictors\\ estimate $\hat L_P$};
\draw[->, thick] (mem) -- (ho); \draw[->, thick] (mem) -- (he); \draw[->, thick] (mem) -- (pi);
\draw[->, dashed, black!60] (ho) -- (hop); \draw[->, dashed, black!60] (he) -- (hep); \draw[->, dashed, black!60] (pi) -- (hp);
\end{tikzpicture}
\end{center}
\begin{itemize}\small
\item The memory parameters $\theta$ are trained on $\ell_O + \lambda\,\ell_E$, the two terms of the decomposition $D_t = D^0_t + \dE_t$. Gradients from the policy do not reach the memory ($\sg$ stops them).
\item The dashed networks receive the window in addition. Their losses are subtracted from $\ell_O, \ell_E, \ell_P$ to form the estimates $\hat L_O, \hat L_E, \hat L_P$; they do not contribute gradients to $\theta$.
\item In population, with networks that have learned the true laws, $\E[\ell_O + \ell_E] = \text{const} + D_t + I(h_t; A_t \mid M_t)$ on the training data: the memory objective is the state defect plus a term that is zero at the memory that generated the data. This is the same objective as action-conditioned prediction of $O_{t+1}$; the decomposition identifies its two parts and provides the estimates.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Summary}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Established}
\begin{itemize}
\item $D_t = 0$ implies that the Bellman recursion on memory is exact; with every action tried, it gives $Q^*$.
\item $\dE_t = I(h_t; A_t \mid M_t, O_{t+1}) \ge 0$ and $\dP_t = I(h_t; A_{t+1} \mid M_t, A_t) \ge 0$.
\item $D_t = D^0_t + \dE_t$ and $0 \le \dP_t \le D_t$.
\item The window versions satisfy the same identities; the difference from $D_t$ is the information older than the window.
\item Action-conditioned prediction already minimises $D_t$ on the data; the proposed memory loss coincides with it up to an inertia term.
\end{itemize}
\column{0.5\textwidth}
\textbf{Open}
\begin{itemize}
\item Whether gradient training reaches $D_t = 0$.
\item Whether the estimates $\hat L_O, \hat L_E, \hat L_P$ track the true gaps with finite data and imperfect networks.
\item Whether the inverse and next-action predictors improve learning relative to a single action-conditioned predictor. This can be tested on small environments where both parts of the defect are known exactly.
\item Recovery time after the environment changes.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Backup: matching pennies in numbers}
Coin $C_t$ fair, action $A_t$ uniform, reward $R_{t+1} = 1$ if $A_t = C_t$, observation $O_{t+1} = (C_{t+1}, R_{t+1})$, constant memory $M_t = 0$.
\vspace{2mm}
\begin{columns}[T]\small
\column{0.5\textwidth}
\textbf{Prediction without the action}
\begin{align*}
H(O_{t+1} \mid M_t) &= H(C_{t+1}) + H(R_{t+1}) = 1 + 1 \\
H(O_{t+1} \mid h_t) &= H(C_{t+1}) + H(R_{t+1} \mid C_t) = 1 + 1
\end{align*}
$R_{t+1}$ is a fair coin even given $C_t$, because $A_t$ is uniform. So $D^0_t = I(h_t; O_{t+1} \mid M_t) = 0$.
\column{0.5\textwidth}
\textbf{Prediction with the action}
\begin{align*}
H(O_{t+1} \mid M_t, A_t) &= 1 + H(R_{t+1} \mid A_t) = 1 + 1 \\
H(O_{t+1} \mid h_t, A_t) &= 1 + H(R_{t+1} \mid C_t, A_t) = 1 + 0
\end{align*}
Given the action and the coin, the reward is determined. So $D_t = I(h_t; O_{t+1} \mid M_t, A_t) = 1$ bit.
\end{columns}
\vspace{3mm}
Empowerment: $E_t = H(O_{t+1} \mid h_t) - H(O_{t+1} \mid h_t, A_t) = 2 - 1 = 1$ bit at the history; $E_t(M) = 2 - 2 = 0$ at the memory; $\dE_t = 1$ bit $= D_t - D^0_t$. Plasticity mismatch: $\dP_t = 0$, since the uniform policy ignores its memory. With $M_t = C_t$ all defects vanish.
\end{frame}
