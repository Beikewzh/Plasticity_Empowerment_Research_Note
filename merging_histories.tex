\begin{abstract}
An agent with a memory of fixed size gives the same action distribution to all histories that have the same memory value. This note derives, one layer at a time, what this costs in return. It now contains the setting and Layer~1. Each layer states its question, the assumptions it adds, its derivation line by line, and its conclusion.
\end{abstract}

\section{Setting}\label{sec:setting}

The history is $h_t=(O_1,A_1,\dots,A_{t-1},O_t)$, and $h$ denotes a value of $h_t$. An observation $o$ contains a reward $r(o)$, and $R_{t+1}=r(O_{t+1})$. All observation and action sets are finite. A final time $T$ and a discount factor $\gamma\in[0,1]$ are fixed, and actions are taken at $t=1,\dots,T-1$.

\begin{assumption}[Environment]\label{as:env}
$O_1$ has a fixed law, and $O_{t+1}\sim e(\cdot\mid h_t,A_t)$ for a fixed $e$. The environment may depend on the whole history.
\end{assumption}

\begin{assumption}[Memory]\label{as:mem}
$M_1=f_1(O_1)$ and $M_{t+1}=f(M_t,A_t,O_{t+1})$ for fixed maps $f_1,f$. Hence $M_t$ is a function of $h_t$, and $m$ denotes a value of $M_t$.
\end{assumption}

\begin{assumption}[Policy]\label{as:pol}
$P(A_t=a\mid h_t)=\pi(a\mid M_t)$: the action depends on the history only through the memory.
\end{assumption}

\begin{definition}[Agents and laws]\label{def:agent}
An agent is a function $\lambda(a\mid h)$ that gives a distribution over actions at every history. Our agent is $\lambda(a\mid h)=\pi(a\mid m)$, with $m$ the memory value of $h$. $P_\lambda$ is the law of $(O_1,A_1,O_2,\dots)$ when the actions are drawn from $\lambda$ and the observations from $e$, and $P$ is this law for our agent. Under every $P_\lambda$, $M_t$ is our memory of $h_t$, computed with $f_1$ and $f$. We write $P_\lambda(h\mid m)$ for $P_\lambda(h_t=h\mid M_t=m)$.
\end{definition}

\begin{definition}[Return and values]\label{def:value}
$G_t=R_{t+1}+\gamma\,G_{t+1}$ with $G_T=0$, and $J(\lambda)=\E_\lambda[G_1]$. For every history $h$ and action $a$ at time $t$,
\begin{align}
Q^\pi_t(h,a)&=\sum_o e(o\mid h,a)\,\big[r(o)+\gamma\,V^\pi_{t+1}(h,a,o)\big], \label{eq:bellman}\\
V^\pi_t(h)&=\sum_a\pi(a\mid m)\,Q^\pi_t(h,a),\qquad V^\pi_T(h)=0, \nonumber
\end{align}
where $(h,a,o)$ is the next history and $m$ is the memory value of $h$. So $Q^\pi_t(h,a)$ is the expected return when $a$ is taken after $h$ and our agent acts afterwards. It is defined for every history, whether or not our agent visits it.
\end{definition}

\section{Layer 1: at which memory values return is lost}\label{sec:layer1}

\paragraph{Question.} Our agent collects less return than another agent $\lambda$. At which memory values does this happen, and how much of it comes from giving one action distribution to all the histories with the same memory value?

\paragraph{Assumptions added.} None. The other agent $\lambda$ is arbitrary.

\begin{lemma}[Known: performance difference]\label{lem:pdl}
Under Assumptions~\ref{as:env} to~\ref{as:pol}, for every agent $\lambda$,
\[
J(\lambda)-J(\pi)=\sum_{t=1}^{T-1}\gamma^{t-1}\sum_hP_\lambda(h_t=h)\Big[\sum_a\lambda(a\mid h)\,Q^\pi_t(h,a)-V^\pi_t(h)\Big].
\]
\end{lemma}

\begin{proof}
\citet{kakade2002cpi} prove this for a Markov decision process. Here the history plays the role of the state, so we write the steps.
\begin{align*}
J(\lambda)-J(\pi)
&=\E_\lambda[G_1]-\E_\lambda\big[V^\pi_1(h_1)\big] \why{$h_1=O_1$ has the same law under both agents; $J(\pi)=\E[V^\pi_1(h_1)]$}\\
&=\E_\lambda\Big[\sum_{t=1}^{T-1}\gamma^{t-1}\big(R_{t+1}+\gamma V^\pi_{t+1}(h_{t+1})-V^\pi_t(h_t)\big)\Big] \why{$G_1=\sum_t\gamma^{t-1}R_{t+1}$; the $V^\pi$ terms sum to $-V^\pi_1(h_1)$, as $V^\pi_T=0$}\\
&=\sum_{t=1}^{T-1}\gamma^{t-1}\sum_hP_\lambda(h_t=h)\sum_a\lambda(a\mid h)\big[Q^\pi_t(h,a)-V^\pi_t(h)\big] \why{given $h_t=h$, $A_t=a$: \eqref{eq:bellman}; $A_t\sim\lambda(\cdot\mid h)$}.
\end{align*}
\end{proof}

\begin{definition}[Four averages at a memory value]\label{def:four}
Fix a time $t$, a memory value $m$ and an agent $\lambda$. Over the histories $h$ with memory value $m$, weighted by $P_\lambda(h\mid m)$:
\begin{align}
&\sum_hP_\lambda(h\mid m)\,\max_aQ^\pi_t(h,a) &&\text{every history takes its own best action,} \label{eq:own}\\
&\max_a\sum_hP_\lambda(h\mid m)\,Q^\pi_t(h,a) &&\text{all histories take one action, the best on average,} \label{eq:one}\\
&\sum_a\pi(a\mid m)\sum_hP_\lambda(h\mid m)\,Q^\pi_t(h,a) &&\text{all histories take the action of our policy,} \label{eq:pi}\\
&\sum_hP_\lambda(h\mid m)\sum_a\lambda(a\mid h)\,Q^\pi_t(h,a) &&\text{every history takes the action of $\lambda$.} \label{eq:lam}
\end{align}
\end{definition}

\begin{proposition}[Layer 1]\label{prop:layer1}
Under Assumptions~\ref{as:env} to~\ref{as:pol}, for every agent $\lambda$,
\[
J(\lambda)-J(\pi)=\sum_{t=1}^{T-1}\gamma^{t-1}\sum_mP_\lambda(M_t=m)\Big[\big(\eqref{eq:own}-\eqref{eq:one}\big)+\big(\eqref{eq:one}-\eqref{eq:pi}\big)-\big(\eqref{eq:own}-\eqref{eq:lam}\big)\Big],
\]
and each of the three differences in round brackets is nonnegative. Hence
\[
J(\lambda)-J(\pi)\le\sum_{t=1}^{T-1}\gamma^{t-1}\sum_mP_\lambda(M_t=m)\Big[\big(\eqref{eq:own}-\eqref{eq:one}\big)+\big(\eqref{eq:one}-\eqref{eq:pi}\big)\Big].
\]
\end{proposition}

\begin{proof}
Fix $t$. The inner sum of Lemma~\ref{lem:pdl} is
\begin{align*}
&\sum_hP_\lambda(h_t=h)\Big[\sum_a\lambda(a\mid h)\,Q^\pi_t(h,a)-V^\pi_t(h)\Big]\\
&\quad=\sum_mP_\lambda(M_t=m)\sum_hP_\lambda(h\mid m)\Big[\sum_a\lambda(a\mid h)\,Q^\pi_t(h,a)-\sum_a\pi(a\mid m)\,Q^\pi_t(h,a)\Big] \why{$M_t$ is a function of $h_t$; Definition~\ref{def:value}}\\
&\quad=\sum_mP_\lambda(M_t=m)\big[\eqref{eq:lam}-\eqref{eq:pi}\big] \why{Definition~\ref{def:four}}\\
&\quad=\sum_mP_\lambda(M_t=m)\Big[\big(\eqref{eq:own}-\eqref{eq:one}\big)+\big(\eqref{eq:one}-\eqref{eq:pi}\big)-\big(\eqref{eq:own}-\eqref{eq:lam}\big)\Big] \why{add and subtract \eqref{eq:own} and \eqref{eq:one}}.
\end{align*}
The three differences are nonnegative for the following reasons.
\begin{align*}
\eqref{eq:own}-\eqref{eq:one}&\ge0 &&\text{for every $a$ and $h$, $\max_bQ^\pi_t(h,b)\ge Q^\pi_t(h,a)$; average over $h$, then take the best $a$;}\\
\eqref{eq:one}-\eqref{eq:pi}&\ge0 &&\text{a maximum over $a$ is at least the average over $a$ with weights $\pi(a\mid m)$;}\\
\eqref{eq:own}-\eqref{eq:lam}&\ge0 &&\text{at each $h$, a maximum over $a$ is at least the average with weights $\lambda(a\mid h)$.}
\end{align*}
Dropping the third difference gives the inequality.
\end{proof}

\paragraph{Conclusion.} The difference in return is a sum over times and memory values of three differences. The first, $\eqref{eq:own}-\eqref{eq:one}$, is what is lost at $m$ because all the histories with memory value $m$ must take the same action; it is zero when these histories share a best action. The second is what our policy loses at $m$ against the best single action, and the third is what $\lambda$ loses against the best action at each history.

\paragraph{The agent compared with.} The statement holds for every $\lambda$. In the next layers $\lambda$ is an agent with the largest $J$ among the agents that satisfy Assumptions~\ref{as:mem} and~\ref{as:pol} with as many memory values as ours. The weights $P_\lambda(M_t=m)$ and $P_\lambda(h\mid m)$ are then those of the histories that this agent visits, not of our own.

\paragraph{Known and checked.} Lemma~\ref{lem:pdl} is known. The grouping of its sum by our memory value into three differences was not found in a short search. Both were checked by exact enumeration for 1{,}776 pairs of our agent and an agent $\lambda$, in small random environments whose next observation depends on the whole history and in a switching bandit (\texttt{merging\_histories/checks/lambda\_path.py} in the repository \texttt{crl-theory-notes}): largest error $1.3\times10^{-15}$, and no negative difference at 21{,}360 pairs $(t,m)$.
